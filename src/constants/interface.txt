export interface IUser {
  id: number | string;
  email: string;
  fullname: string;
  phone: string;
  token: string;
  warehouse?: string;
  district?: string;
  province?: string;
  addressDetail?: string;
  role: string;
  balance: number;
  transferContent?: string;
  avatar?: string;
}

export interface IProduct {
  id: number | string;
  link_product: string;
  color?: string;
  size?: string;
  number: number;
  price: number; // giá theo tệ
  note?: string;
  image?: string;
}

export type IOrderProduct = IProduct & {
  qtyOrder: number;
  qtyReceive: number;
  staffNote?: string;
};
export interface IOrder {
  id: number | string;
  code: number | string;
  status: number; // trạng thái đơn
  images?: string; // ảnh đơn
  shippingType: string; // phương thức vận chuyển
  products: Array<IOrderProduct>; // danh sách sản phẩm
  noteForOwner: string; // ghi chú cho người tạo
  noteForStaff: string; // ghi chú cho staff
  orderFeePercent: number; // % phí mua hàng
  exchangeRate: number; // tỉ giá
  stockWarehouse: string | number; // Hàng về kho
  userWarehouse: string | number; // Kho khách chọn
  extraFee?: number; // phí trả thêm
  shippingCnFee?: number; // phí ship nội địa TQ
  tallyFee?: number; // phí kiểm đếm
  itemTotalCost: number;
  totalFee: number; // tổng phí
  totalPaid: number; // đã trả

  depositAt?: string; // Thời gian đặt cọc
  purchasedAt?: string; // Thời gian mua (bên order mua)
  transitToVnAt?: string; // Trên đường về VN
  stockInVnAt?: string; // Trong kho VN
  returnAt?: string; // Thời gian trả hàng
  sellerShippedAt?: string; // thời gian người bán giao

  weightBasedFee?: number; // tiền cân thực
  volumnBasedFee?: number; // tiền cân quy đổi từ thể tích
  totalWeightFee?: number; // tiền cân tính
  woodPackagingFee?: number; // phí đóng gỗ
  isWoodPackaging?: boolean; // đóng gỗ?
  extraShippingFee?: number; // Cước vận phát sinh
  weight?: number; // cân nặng
  weightBasedVolumn?: number; // cân nặng theo thể tích
  weightRate?: number; // dơn giá cân nặng

  modified_time?: string | number;
  created_time?: string | number;
}
